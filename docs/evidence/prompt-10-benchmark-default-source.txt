/** Sample only this dedicated browser's process tree. Sum RSS is NOT unique heap
 * or WASM usage: shared pages may be counted twice, brief peaks may be missed. */
function memorySampler(pid: number | undefined) {
  const samples: { elapsedMs: number; rssBytes: number; processes: number }[] = [];
  const began = performance.now(); let unavailable: string | null = null;
  const sample = () => {
    if (!pid || !['darwin', 'linux'].includes(platform())) { unavailable = 'Process-tree RSS sampling is unavailable on this platform'; return; }
    try {
      const rows = execFileSync('ps', ['-axo', 'pid=,ppid=,rss='], { encoding: 'utf8', timeout: 2000 })
        .trim().split('\n').map(row => row.trim().split(/\s+/).map(Number));
      const ids = new Set([pid]); let previous = 0;
      while (previous !== ids.size) { previous = ids.size; for (const [id, parent] of rows) if (parent && id && ids.has(parent)) ids.add(id); }
      const own = rows.filter(([id]) => id && ids.has(id));
      samples.push({ elapsedMs: round(performance.now() - began), rssBytes: own.reduce((sum, row) => sum + (row[2] ?? 0) * 1024, 0), processes: own.length });
    } catch { unavailable = 'Host process sampling was denied or timed out; no portable total-browser memory API was substituted'; }
  };
  sample(); const timer = setInterval(sample, 500);
  return () => { clearInterval(timer); sample(); return { method: '500 ms sampled sum of dedicated browser process-tree RSS, shared pages not deduplicated',
    sampledPeakBytes: samples.length ? Math.max(...samples.map(s => s.rssBytes)) : null, unavailable, samples }; };
}

async function select(page: Page, demo: typeof demos[number]) {
  await page.goto('/#/inspect');
  await page.getByLabel('Choose a local file', { exact: true }).setInputFiles(`tests/fixtures/dos/${demo.file}`);
  const at = performance.now(); await page.getByRole('button', { name: 'Analyze file', exact: true }).click();
  await expect(page.getByRole('radio', { name: `Choose ${demo.entry} as entrypoint`, exact: true })).toBeVisible();
  const inspectMs = performance.now() - at;
  await page.getByRole('radio', { name: `Choose ${demo.entry} as entrypoint`, exact: true }).check();
  await page.getByRole('link', { name: 'View compatibility', exact: true }).click();
  await page.getByRole('link', { name: 'Configure launch' }).click();
  await page.getByRole('button', { name: 'Use these settings and continue' }).click();
  await expect(page.getByRole('button', { name: 'Start DOS program', exact: true })).toBeEnabled();
  return inspectMs;
}
async function launch(page: Page, demo: typeof demos[number]) {
  const at = performance.now(); await start(page); const firstFrameMs = performance.now() - at;
  await command(page, demo.instruction); const output = await downloadGuest(page, demo.output);
  if (demo.file === 'MEM.EXE') assert.match(output.toString(), /Conventional[\s\S]*Extended \(XMS\)/);
  else assert.equal(output.toString(), 'BENCH OK\r\n');
  const confirmedOutputMs = performance.now() - at, stopAt = performance.now();
  await page.getByRole('button', { name: 'Stop session' }).click();
  await expect(page.locator('iframe')).toHaveCount(0); await expect.poll(() => page.workers().length).toBe(0);
  const stopMs = performance.now() - stopAt; assert(stopMs < 2000);
  return { firstFrameMs: round(firstFrameMs), confirmedOutputMs: round(confirmedOutputMs), stopMs: round(stopMs), outputBytes: output.length, outputSha256: sha(output) };
}
